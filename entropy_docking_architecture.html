4.1 Ear-Canal Heart Rate → SCI Pipeline

// Swift — AirPods Pro 3 Heart Rate → SCI
import HealthKit

class AirPodsPhysiologicalEngine {
  let store = HKHealthStore()
  var rrIntervals: [Double] = []
  let windowSize = 30 // 30 R-R intervals for SCI

  // Shannon entropy of R-R interval distribution
  func computeSCI() -> Double {
    guard rrIntervals.count >= windowSize else { return 0 }
    let window = Array(rrIntervals.suffix(windowSize))

    // Bin R-R intervals (50ms bins, 10-30 range)
    var histogram = [Int](repeating: 0, count: 20)
    for rr in window {
      let bin = min(19, max(0, Int((rr - 0.5) / 0.05)))
      histogram[bin] += 1
    }

    // Shannon entropy H = -Σ p_i log2(p_i)
    var H: Double = 0
    for count in histogram where count > 0 {
      let p = Double(count) / Double(windowSize)
      H -= p * log2(p)
    }

    // SCI = 1 - H/H_max (collapse = low entropy)
    let H_max = log2(Double(windowSize))
    return 1.0 - H / H_max
  }

  // Physiological temperature from heart rate
  var physiologicalTemp: Double {
    // HR 60 → 280K (rest), HR 120 → 320K (active)
    let hr = 60.0 / (rrIntervals.last ?? 1.0)
    return 280.0 + (hr - 60) * (40.0 / 60.0)
  }
}

// Swift — MusicKit real-time BPM + audio analysis
import MusicKit
import AVFAudio

class MusicTempoEngine: ObservableObject {
  @Published var currentBPM: Double = 120
  @Published var deltaBPM: Double = 0
  @Published var audioEntropy: Double = 0
  @Published var trackName: String = ""
  @Published var trackArtist: String = ""

  private let engine = AVAudioEngine()
  private var prevBPM: Double = 120
  private var fftBuffer: [Float] = []

  func startMonitoring() async {
    // 1. MusicKit now-playing observation
    for await entry in MusicPlayer.shared.queue.currentEntry.values {
      if let song = entry?.item as? Song {
        trackName = song.title
        trackArtist = song.artistName
      }
    }
  }

  func startAudioTap() {
    // Tap system audio output for real-time analysis
    let input = engine.inputNode
    let format = input.outputFormat(forBus: 0)

    input.installTap(onBus: 0, bufferSize: 4096,
      format: format) { [weak self] buffer, time in
      self?.processAudioBuffer(buffer)
    }
    try? engine.start()
  }

  private func processAudioBuffer(_ buffer: AVAudioPCMBuffer) {
    guard let data = buffer.floatChannelData?[0] else { return }
    let count = Int(buffer.frameLength)

    // Onset detection for BPM
    var energy: Float = 0
    for i in 0..<count { energy += data[i] * data[i] }
    energy /= Float(count)

    // Shannon entropy of spectral distribution
    var histogram = [Float](repeating: 0, count: 32)
    for i in 0..<count {
      let bin = min(31, Int((data[i] + 1) * 16))
      histogram[bin] += 1
    }
    var H: Float = 0
    for h in histogram where h > 0 {
      let p = h / Float(count)
      H -= p * log2(p)
    }
    audioEntropy = Double(H)
  }
}
